// ============================================================ // KTIS - Hopper Feed System // Google Apps Script // ============================================================ const SHEET_ID = '1oD-5HPoIF2UO0mIsCPAHKuHXlX9_yDx769M4PGjN69g'; const SHEET_NAME = 'Hopper'; // ============================================================ // ฟังก์ชันสำหรับส่ง JSON / JSONP // ============================================================ function jsonResponse(data, callback) { const json = JSON.stringify(data); // ถ้ามี callback ให้ตอบกลับเป็น JSONP if (callback) { // ป้องกัน callback ที่ไม่ใช่ชื่อฟังก์ชัน JavaScript ที่เหมาะสม if (!/^[a-zA-Z_$][0-9a-zA-Z_$]*(\.[a-zA-Z_$][0-9a-zA-Z_$]*)*$/.test(callback)) { return ContentService .createTextOutput(JSON.stringify({ result: 'error', message: 'Invalid callback' })) .setMimeType(ContentService.MimeType.JSON); } return ContentService .createTextOutput(callback + '(' + json + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } // ถ้าไม่มี callback ให้ตอบ JSON ปกติ return ContentService .createTextOutput(json) .setMimeType(ContentService.MimeType.JSON); } // ============================================================ // doGet // ใช้สำหรับดึงข้อมูลจาก Google Sheet // ============================================================ function doGet(e) { try { const ss = SpreadsheetApp.openById(SHEET_ID); const sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { return jsonResponse({ result: 'error', message: 'ไม่พบ Sheet ชื่อ ' + SHEET_NAME }, e && e.parameter ? e.parameter.callback : null); } // ======================================================== // 1. ดึงตัวเลขก่อนฟีด A / B // ======================================================== if (e.parameter.action === 'get_last_readings') { const lastRow = sheet.getLastRow(); let preFeedA = '00000000'; let preFeedB = '00000000'; if (lastRow > 1) { const lastRowData = sheet .getRange(lastRow, 1, 1, sheet.getLastColumn()) .getValues()[0]; // E = หลังฟีด A const postFeedAFromSheet = lastRowData[4]; // F = หลังฟีด B const postFeedBFromSheet = lastRowData[5]; if ( postFeedAFromSheet !== undefined && postFeedAFromSheet !== null && String(postFeedAFromSheet).trim() !== '' ) { let valueA = String(postFeedAFromSheet).trim(); // กรณี Google Sheet เก็บเป็นตัวเลข if (!isNaN(Number(valueA))) { valueA = String(Math.round(Number(valueA))); } preFeedA = valueA.padStart(8, '0'); } if ( postFeedBFromSheet !== undefined && postFeedBFromSheet !== null && String(postFeedBFromSheet).trim() !== '' ) { let valueB = String(postFeedBFromSheet).trim(); // กรณี Google Sheet เก็บเป็นตัวเลข if (!isNaN(Number(valueB))) { valueB = String(Math.round(Number(valueB))); } preFeedB = valueB.padStart(8, '0'); } } Logger.log( 'Sending last readings: preFeedA=' + preFeedA + ', preFeedB=' + preFeedB ); return jsonResponse( { preFeedA: preFeedA, preFeedB: preFeedB }, e.parameter.callback ); } // ======================================================== // 2. รายงานสรุป // ======================================================== else if (e.parameter.action === 'get_report') { const startDateParam = e.parameter.startDate; const endDateParam = e.parameter.endDate; const statusFilterParam = e.parameter.feedStatusFilter; const allData = sheet.getDataRange().getValues(); if (allData.length <= 1) { Logger.log( 'No data found in sheet or only header row exists for report.' ); return jsonResponse( { summary: { totalTonA: '0.000', totalTonB: '0.000', grandTotalTon: '0.000', normalCount: 0, abnormalCount: 0, filteredRecordsCount: 0 } }, e.parameter.callback ); } const records = allData.slice(1); let filteredRecords = records; // ------------------------------------------------------ // กรองตามวันที่ // ------------------------------------------------------ if (startDateParam && endDateParam) { const startDate = new Date(startDateParam + 'T00:00:00'); const endDate = new Date(endDateParam + 'T23:59:59'); filteredRecords = filteredRecords.filter(row => { const rowDateValue = row[0]; if (!rowDateValue) { return false; } let sheetDate; if (rowDateValue instanceof Date) { sheetDate = new Date( rowDateValue.getFullYear(), rowDateValue.getMonth(), rowDateValue.getDate() ); } else { const dateStr = String(rowDateValue).trim(); const dateParts = dateStr.split('/'); if (dateParts.length === 3) { sheetDate = new Date( `${dateParts[2]}-${dateParts[1]}-${dateParts[0]}T00:00:00` ); } else { return false; } } if (isNaN(sheetDate.getTime())) { Logger.log( 'Invalid date in sheet row for filtering: ' + rowDateValue ); return false; } return ( sheetDate >= startDate && sheetDate <= endDate ); }); } // ------------------------------------------------------ // กรองสถานะ // ------------------------------------------------------ if ( statusFilterParam && statusFilterParam !== 'ทั้งหมด' ) { filteredRecords = filteredRecords.filter(row => { const feedStatus = String(row[15]).trim(); return feedStatus === statusFilterParam; }); } let totalTonA = 0; let totalTonB = 0; let normalCount = 0; let abnormalCount = 0; filteredRecords.forEach(row => { const quantityA = parseFloat(row[6]) || 0; const quantityB = parseFloat(row[7]) || 0; const feedStatus = String(row[15]).trim(); totalTonA += quantityA; totalTonB += quantityB; if (feedStatus === 'ปกติ') { normalCount++; } else if (feedStatus === 'ไม่ปกติ') { abnormalCount++; } }); const grandTotalTon = totalTonA + totalTonB; const reportSummary = { totalTonA: totalTonA.toFixed(3), totalTonB: totalTonB.toFixed(3), grandTotalTon: grandTotalTon.toFixed(3), normalCount: normalCount, abnormalCount: abnormalCount, filteredRecordsCount: filteredRecords.length }; Logger.log( 'Report Summary to be returned: ' + JSON.stringify(reportSummary) ); return jsonResponse( { summary: reportSummary }, e.parameter.callback ); } // ======================================================== // 3. ดึงข้อมูลหลายรายการสำหรับวันที่เลือก // ======================================================== else if ( e.parameter.action === 'get_records_for_date' ) { const targetDateParam = e.parameter.targetDate; const allData = sheet.getDataRange().getValues(); if (allData.length <= 1) { Logger.log( 'No data found in sheet for get_records_for_date.' ); return jsonResponse( { records: [], totalRecords: 0 }, e.parameter.callback ); } const records = allData.slice(1); const targetDateObj = new Date(targetDateParam + 'T00:00:00'); const matchingRecords = records.filter(row => { const rowDateValue = row[0]; if (!rowDateValue) { return false; } let sheetDate; if (rowDateValue instanceof Date) { sheetDate = new Date( rowDateValue.getFullYear(), rowDateValue.getMonth(), rowDateValue.getDate() ); } else { const dateStr = String(rowDateValue).trim(); if (dateStr.includes('/')) { const parts = dateStr.split('/'); sheetDate = new Date( `${parts[2]}-${parts[1]}-${parts[0]}T00:00:00` ); } else { sheetDate = new Date( dateStr + 'T00:00:00' ); } } if (isNaN(sheetDate.getTime())) { Logger.log( 'Invalid date in sheet row for filtering (get_records_for_date): ' + rowDateValue ); return false; } return ( sheetDate.getTime() === targetDateObj.getTime() ); }); // ------------------------------------------------------ // เรียงตาม DCS สั่งตัก // ------------------------------------------------------ matchingRecords.sort((a, b) => { const timeA = a[9] instanceof Date ? a[9] : new Date( `1899-12-30T${String(a[9] || '00:00')}:00` ); const timeB = b[9] instanceof Date ? b[9] : new Date( `1899-12-30T${String(b[9] || '00:00')}:00` ); return ( timeA.getTime() - timeB.getTime() ); }); // ------------------------------------------------------ // จัดรูปข้อมูล // ------------------------------------------------------ const formattedRecords = matchingRecords.map(selectedRecord => { return { date: selectedRecord[0] instanceof Date ? Utilities.formatDate( selectedRecord[0], ss.getSpreadsheetTimeZone(), 'dd/MM/yyyy' ) : selectedRecord[0], recorderEmployee: selectedRecord[1], preFeedA: selectedRecord[2], preFeedB: selectedRecord[3], postFeedA: selectedRecord[4], postFeedB: selectedRecord[5], quantityA: selectedRecord[6], quantityB: selectedRecord[7], totalQuantity: selectedRecord[8], dcsOrderTime: selectedRecord[9] ? Utilities.formatDate( selectedRecord[9], ss.getSpreadsheetTimeZone(), 'HH:mm' ) : '', dcsStopTime: selectedRecord[10] ? Utilities.formatDate( selectedRecord[10], ss.getSpreadsheetTimeZone(), 'HH:mm' ) : '', hopperSugarA: selectedRecord[11], hopperSugarB: selectedRecord[12], warehouseEmployee: selectedRecord[13], lsEmployeeCheck: selectedRecord[14], feedStatus: selectedRecord[15], notes: selectedRecord[16], checkerEmployee: selectedRecord[17] }; }); Logger.log( 'Returning ' + formattedRecords.length + ' records for date ' + targetDateParam ); return jsonResponse( { records: formattedRecords, totalRecords: formattedRecords.length }, e.parameter.callback ); } // ======================================================== // กรณีไม่มี action // ======================================================== return jsonResponse( { result: 'success', message: 'KTIS Hopper Feed Web App is running.' }, e.parameter.callback ); } catch (error) { Logger.log( 'doGet Error: ' + error.message ); return jsonResponse( { result: 'error', message: error.message }, e && e.parameter ? e.parameter.callback : null ); } } // ============================================================ // doPost // ใช้สำหรับบันทึกข้อมูลลง Google Sheet // ============================================================ function doPost(e) { try { const ss = SpreadsheetApp.openById(SHEET_ID); const sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { throw new Error( 'ไม่พบ Sheet ชื่อ ' + SHEET_NAME ); } Logger.log( 'ข้อมูลที่ได้รับจาก Web App (e.parameter): ' + JSON.stringify(e.parameter) ); const data = e.parameter; // -------------------------------------------------------- // แปลงวันที่จาก YYYY-MM-DD เป็น DD/MM/YYYY // -------------------------------------------------------- let formattedDate = ''; if (data.date) { const parts = data.date.split('-'); if (parts.length === 3) { formattedDate = `${parts[2]}/${parts[1]}/${parts[0]}`; } else { formattedDate = data.date; } } // -------------------------------------------------------- // เตรียมข้อมูลลง Sheet // -------------------------------------------------------- const row = [ formattedDate, // A วันที่ data.recorderEmployee, // B ผู้บันทึก data.preFeedA, // C ก่อนฟีด A data.preFeedB, // D ก่อนฟีด B data.postFeedA, // E หลังฟีด A data.postFeedB, // F หลังฟีด B data.quantityA, // G Ton A data.quantityB, // H Ton B data.totalQuantity, // I รวม data.dcsOrderTime, // J DCS สั่งตัก data.dcsStopTime, // K DCS สั่งหยุด data.hopperSugarA, // L น้ำตาล Hopper A data.hopperSugarB, // M น้ำตาล Hopper B data.warehouseEmployee, // N พนักงานคลัง data.lsEmployeeCheck, // O พนักงาน LS data.feedStatus, // P สถานะ data.notes, // Q หมายเหตุ data.checkerEmployee // R ผู้ตรวจสอบ ]; Logger.log( 'แถวข้อมูลที่จะบันทึก: ' + JSON.stringify(row) ); sheet.appendRow(row); return ContentService .createTextOutput( JSON.stringify({ result: 'success' }) ) .setMimeType( ContentService.MimeType.JSON ); } catch (error) { Logger.log( 'เกิดข้อผิดพลาดในการบันทึกข้อมูล: ' + error.message ); return ContentService .createTextOutput( JSON.stringify({ result: 'error', message: error.message }) ) .setMimeType( ContentService.MimeType.JSON ); } }
